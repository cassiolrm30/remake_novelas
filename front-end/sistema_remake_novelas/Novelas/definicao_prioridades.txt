	<script>
		var qtLinhas = 1;
		var tabela   = document.getElementById("tblPersonagens");
        var rotulos  = ["Personagem", "Gênero", "Etnia", "Faixa Etária", "Peso", "Altura"];
        var larguras = ["50", "10", "10", "10", "10", "10"];
        var aligns   = ["left", "center", "center", "center", "center", "center"];

		function montarTabela()
		{
			$("#tblPersonagens tr").remove(); // reinicializar a tabela
			var linha = tabela.insertRow(0);
			// Definindo o título da cada coluna da tabela
			for (var i = 0; i < rotulos.length; i++)
			{
				var celulaTitulo = linha.insertCell(i);
				celulaTitulo.id = "T" + (i+1);
				celulaTitulo.className = "celulaTitulo";
				celulaTitulo.innerHTML = rotulos[i];
				celulaTitulo.style.width = larguras[i];
				celulaTitulo.style.textAlign = aligns[i];
				if (celulaTitulo.style.textAlign == "left")
					celulaTitulo.style.paddingLeft = "5px";
				if (celulaTitulo.style.textAlign == "right")
					celulaTitulo.style.paddingRight = "5px";
			}
			// Definindo o corpo da tabela
			var linha = [];
			adicionarLinha(linha);
		}

		function adicionarLinha(linhas)
		{
			var linha = tabela.insertRow(qtLinhas);
			var conteudo = linhas;
			for (var i = 0; i < conteudo.length; i++)
			{
				var dadosConteudo = conteudo[i].split('|');
				var celula = linha.insertCell(i);
				celula.id = "C" + qtLinhas + (i+1);
				celula.className = (i == 0 ? "coluna1" : "celulaConteudo");
				celula.innerHTML = dadosConteudo[0];
				celula.style.textAlign = dadosConteudo[1];
				if (celula.style.textAlign == "left")
					celula.style.paddingLeft = "5px";
				if (celula.style.textAlign == "right")
					celula.style.paddingRight = "5px";
			}
			qtLinhas++;
		}

        function montarComboNovelas()
		{
			let data = getNovelasComPersonagens();
            for (var i = 0; i < data.length; i++)
			{
                var option   = document.createElement("option");
                option.value = data[i].id;
                option.text  = data[i].titulo;
                document.getElementById("cboNovela").add(option);
            }
        }

		function getPrioridades(novelaId)
		{
			var personagemIds = "";
            var data = personagens;
			montarTabela();
			for (var i = 0; i < data.length; i++)
			{
				var linha = [];
				linha.push(data[i].nome + "|left");
                linha.push("<input type='checkbox' id='chk" + data[i].id + "-1' value='" + data[i].generoId + "' " + (data[i].generoObrig == "1" ? " checked" : "") + ">|center");
                linha.push("<input type='checkbox' id='chk" + data[i].id + "-2' value='" + data[i].etniaId + "' " + (data[i].etniaObrig == "1" ? " checked" : "") + ">|center");
                linha.push("<input type='checkbox' id='chk" + data[i].id + "-3' value='" + data[i].faixaEtariaId + "' " + (data[i].faixaEtariaObrig == "1" ? " checked" : "") + ">|center");
                linha.push("<input type='checkbox' id='chk" + data[i].id + "-4' value='" + data[i].faixaPesoId + "' " + (data[i].faixaPesoObrig == "1" ? " checked" : "") + ">|center");
                linha.push("<input type='checkbox' id='chk" + data[i].id + "-5' value='" + data[i].faixaEstaturaId + "' " + (data[i].faixaEstaturaObrig == "1" ? " checked" : "") + ">|center");
				adicionarLinha(linha);
                personagemIds += data[i].PersonagemId + "|";
			}
			$("#PersonagemIds").val(personagemIds);
		}

		function setPrioridades()
		{
			if ($("#cboNovela").val() == "")
			{
				alert("É necessário preencher a novela.");
				$("#cboNovela").focus();
				return;
			}
			var qtSelecionados = $("input:checked").length;
			if (qtSelecionados == 0)
			{
				alert("É necessário selecionar pelo menos uma caixa de seleção.");
				return;
			}

			var qtColunas	      = rotulos.length;
			var celulas			  = $("#tblPersonagens td[class!='celulaTitulo']");
			var qtLinhas		  = (celulas.length / rotulos.length);
			var prioridadesPerfis = [];
			var indice			  = 0;
			for (var i = 0; i < qtLinhas; i++)
			{
                var v_ids = $("#PersonagemIds").val().split('|');
                var id = v_ids[i];
                var generoId 		  = $("#chk"+id+"-1").val();
                var etniaId			  = $("#chk"+id+"-2").val();
                var faixaEtariaId	  = $("#chk"+id+"-3").val();
                var faixaPesoId		  = $("#chk"+id+"-4").val();
                var faixaEstaturaId   = $("#chk"+id+"-5").val();
				var flagGenero 		  = $("#chk"+id+"-1").prop("checked") ? "true" : "false";
				var flagEtnia 		  = $("#chk"+id+"-2").prop("checked") ? "true" : "false";
				var flagFaixaEtaria	  = $("#chk"+id+"-3").prop("checked") ? "true" : "false";
                var flagFaixaPeso     = $("#chk"+id+"-4").prop("checked") ? "true" : "false";
                var flagfaixaEstatura = $("#chk"+id+"-5").prop("checked") ? "true" : "false";
                prioridadesPerfis.push ({
										 "Id": id,
										 "NovelaId":		$("#cboNovela").val(),  "Nome":				  celulas[indice].innerText,
										 "GeneroId":		generoId,  				"GeneroObrig":		  flagGenero,
										 "EtniaId":			etniaId,				"EtniaObrig":		  flagEtnia,
										 "FaixaEtariaId":	faixaEtariaId,			"FaixaEtariaObrig":	  flagFaixaEtaria,
										 "FaixaPesoId":		faixaPesoId,			"FaixaPesoObrig":	  flagFaixaPeso,
										 "FaixaEstaturaId": faixaEstaturaId,		"FaixaEstaturaObrig": flagfaixaEstatura
									    });
				indice += qtColunas;
                console.log(prioridadesPerfis[i]);
			}
			
			var dados = prioridadesPerfis;
			$.ajax({
                type: "PUT",
                contentType: "application/json",
                url: URL + "personagem/set_prioridades_perfis/",
                data: JSON.stringify(dados),
				success: function (data, status)
				{
                    if (status != "success")
                        alert("Não foi possível a movimentação!");
					else
					{
                        alert("Operação realizada com sucesso.");
                        return true;
                    }
                },
				error: function ()
				{
                    alert("Erro interno");
                    return false;
                }
            });
		}

		$("#btnSalvar").click(function () { setPrioridades(); });

		$("#btnCancelar").click(function () { window.location.href = "../Index.html"; });

        $("#cboNovela").change(function () { qtLinhas = 1; getPrioridades($(this).val()); });

		$(document).ready(function()
		{
            $("h3.titulo").html("Definição de Prioridades");
			montarComboNovelas();
		});
