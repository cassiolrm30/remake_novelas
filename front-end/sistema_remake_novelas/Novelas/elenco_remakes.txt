        function defineTabelaElenco(novelaId)
        {
            if (novelaId == "")
            {
                $("#btnSalvar").attr("disabled", true);
                $("#btnComparar").attr("disabled", true);
                listagemElenco.Carga = [];
            }
            else
            {
                $("#btnSalvar").attr("disabled", false);
                $("#btnComparar").attr("disabled", false);
                listagemElenco.Carga = getElenco(novelaId);
            }
            listagemElenco.Tabela = document.getElementById("tblElenco");
            listagemElenco.Rotulos = [];
            listagemElenco.Larguras = [];
            listagemElenco.Alinhamentos = [];
            listagemElenco.Inputs = [];
            listagemElenco.Rotulos.push("Personagem");
            listagemElenco.Alinhamentos.push("left");
            for (var i = 0; i < listaIdsVersoes.length; i++)
            {
                listagemElenco.Rotulos.push((i + 1) + "ª Versão" + (i == 0 ? " (Original)" : ""));
                listagemElenco.Alinhamentos.push("center");
            }
            listagemElenco.Rotulos.push("");
            switch (listaIdsVersoes.length)
            {
                case 3: listagemElenco.Larguras = [24, 25, 25, 25, 1]; break;
                case 4: listagemElenco.Larguras = [19, 20, 20, 20, 20, 1]; break;
                default: listagemElenco.Larguras = [33, 33, 33, 1]; break;

		function getElenco(novelaId)
		{
			let resultado = [];
			let personagensDaNovela = [];
			let _personagens = personagens.filter((p) => p.novelaId == novelaId);
			let _atores = atores.sort((a, b) => { if (a.nomeArtistico <= b.nomeArtistico) { return -1; }});
			listaIdsVersoes = ["11", "12"];
		
			for (let i = 0; i < _personagens.length; i++)
			{
				let combosAtores = [];
				debugger;
				for (let j = 0; j < listaIdsVersoes.length; j++)
				{
					let dadosAtores2 = _atores;
					let ID = _personagens[i].id + "-" + listaIdsVersoes[j].id;
					let conteudoCombo = "";
					let v_disabled = "";
					let nomeClasse = "comboHabilitada";
					let linkImagem = "";
					let _interpretacao = interpretacoes.filter((x) => x.personagemId == _personagens[i].id && x.versaoNovelaId == listaIdsVersoes[j].id);
					let _atorId = (_interpretacao == null || _interpretacao.length == 0 ? "0" : _interpretacao[0].atorId);
					if (dadosAtores2.length == 0 || listaIdsVersoes[j].elencoCompleto)
					{
						v_disabled = " disabled";
						nomeClasse = "comboDesabilitada";
						linkImagem += "<span style='padding-left:26px;'>";
					}
					else
					{
						linkImagem += "<span style='padding-left:10px;' id='spanAtor" + ID + "'>";
						linkImagem += "<a href='#' onclick='selecionarComboAtores(" + _personagens[i].id + "," + listaIdsVersoes[j].id + ")';>";
						linkImagem += "<img src='../imagens/SEM_FOTO.jpg' class='modalAtor' data-toggle='modal' data-target='#modalFotoAtor' id='img" + _personagens[i].id + "' />";

					conteudoCombo += "<option value=''>-- Selecione --</option>";

					{
						if (_atorId != "0") { debugger; }
						conteudoCombo += "<option value='" + dadosAtores2[k].id + "'";
						conteudoCombo += (dadosAtores2[k].id == _atorId ? " selected" : "");
						conteudoCombo += ">" + dadosAtores2[k].nomeArtistico + "</option>";
					}
					conteudoCombo += "</select>";
					conteudoCombo += linkImagem;

            {
                alert("É necessário preencher a novela.");
                $("#cboNovela").focus();
                return;
            }
            dadosCadastro.NovelaId = $("#cboNovela").val();
            dadosCadastro.Interpretacoes = [];
            dadosCadastro.VersaoNovelaId = idVersaoAtual;
            var combos = $("select[class='comboHabilitada']");
            for (var i = 0; i < combos.length; i++) {
                if (combos[i].value != "") {
                    var valoresNoID = combos[i].id.replace("cboAtor", "").split('-');
                    var valorDaCombo = combos[i].value;
                    dadosCadastro.Interpretacoes.push
                        ({
                            "PersonagemId": (valoresNoID[0] == "" ? 0 : parseInt(valoresNoID[0])),
                            "VersaoNovelaId": (valoresNoID[1] == "" ? 0 : parseInt(valoresNoID[1])),
                            "AtorId": (valorDaCombo == "" ? 0 : parseInt(valorDaCombo))
                        });
                }
            }

            var interpretacoesVersaoAtual = $("select[id$='-" + idVersaoAtual + "']");
            dadosCadastro.Completo = true;
            for (var i = 0; i < interpretacoesVersaoAtual.length; i++) {
                if (interpretacoesVersaoAtual[i].value == "") {
                    dadosCadastro.Completo = false;
                    break;
                }
            }
            $.ajax(
                {
                    type: "POST",
                    contentType: "application/json",
                    url: URL + "personagem/set_interpretacoes/",
                    data: JSON.stringify(dadosCadastro),
                    async: false,
                    success: function (data) {
                        if (!data.sucesso)
                            alert("Erro interno - não foi possível o cadastro!");
                        else
                            alert("Operação realizada com sucesso.");
                    },
                    error: function (e) {
                        console.log(e.message);
                        alert("Erro interno - não foi possível o cadastro!");
                    }
                });
        }
